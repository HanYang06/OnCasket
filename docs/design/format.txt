# park 文件名 = 唯一名（16 或 32 字符），不绑语义；块内槽连续、不跨 park（004）
# park 定长 = 每个 park 创建时按预算定下 slot_num * slot_size，此后自身不变；不同 park 可以不同
# 槽位永不摘除：位置即身份，偏移 = 1024 + id*slot_size；删除只把 slot_state 清成 空(0)
#   文件只涨不缩，归还空间交给 GC；没写过的槽不存在，文件长度 = 128 B + slot_live*1024 B
#   slot_live = slot_used + slot_dead + 空闲
# version 只在 park 头：管格式，不管数据（本引擎不做数据版本管理）
# 一条链 = 头槽 → 中间槽* → 结束槽（结束槽最后写 = 提交）；头/数据分界靠 header 里的计数
# slot_state 四值：开始 / 中间 / 结束 / 空(全零)。三个常数两两汉明距离 ≥ 14 bit，别改成相邻值
# check 用 XXH3-128（第三方 xxhash，不在 stdlib）；算法：跳过所有 check 字段，其余按盘上顺序全拼
#   —— data 槽算本槽；头槽算本槽再拼上整个 data 区。哈希值必须在写该槽之前算出来（字据）

format_pack type:file
    header                      none                0:1024
        version                 hash32              0:32
        slot_size               int32               32:64
        slot_num                int32               64:96
        slot_live               int32               96:128
        slot_used               int32               128:160
        slot_dead               int32               160:192
        none                    none                192:1024
    slot                        slot                1024:1024+(slot_num*slot_size)
        slot_state              magic32             0:32        # 开始 0x9E3779B1 / 中间 0x85EBCA77 / 结束 0xC2B2AE3D / 空 0
        check                   xxh128              32:160
        # 中间区二选一：slot_header 或 slot_data
        slot_header             none                160:slot_size
            header_slot_num     int32               160:192
            header_slot_end     int32               192:224
            data_slot_num       int32               224:256
            data_slot_end       int32               256:288
            block_body_size     int32               288:320
            block_self_attr_len int32               320:352
            block_self_attr     none                352:8192
                block_id        uuid4               352:480
                none            none                480:8192
        slot_data               none                160:slot_size
            data_body           none                160:8192
